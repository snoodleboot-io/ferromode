## Usage Examples

### Rust: Basic Streaming
```rust
let config = EmdConfig::default();
let predictor = Box::new(ArModel::new(3)?);
let mut decomposer = StreamingDecomposer::new(config, predictor, 4096)?;

for chunk in signal_stream {
    let signal = Signal::from_slice(&chunk)?;
    let result = decomposer.decompose_chunk(&signal)?;
    println!("IMFs: {}", result.imfs.len());
}
```

### Python: Streaming with Metrics
```python
import ferromode_py
import numpy as np

decomposer = ferromode_py.StreamingDecomposer(
    max_imfs=8,
    chunk_size=512,
    boundary_condition="mirror"
)

chunk = np.sin(np.arange(512) * 0.02).astype(np.float64)
result = decomposer.decompose_chunk(chunk)

print(f"IMFs: {result['imfs'].shape[0]}")
print(f"Entropy: {result['metrics']['spectral_entropy']:.3f}")
```
